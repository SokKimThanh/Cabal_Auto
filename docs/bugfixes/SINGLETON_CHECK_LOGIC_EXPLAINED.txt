try:
    # Get reference to global singleton
    instance = monster_editor_module._quick_editor_instance
    
    # Phase 1: Existence check
    instance_exists = instance is not None
    instance_alive = False
    
    # Phase 2: Liveness check
    if instance_exists and instance:
        try:
            # winfo_exists() returns:
            # - 1 (True) if window still exists in memory
            # - 0 (False) if window destroyed
            # - Exception if reference is completely invalid
            instance_alive = bool(instance.winfo_exists())
        except Exception as check_err:
            # Stale reference detected - clean it up
            print(f"Stale instance: {check_err}")
            monster_editor_module._quick_editor_instance = None
            instance_alive = False
    
    # Phase 3: Log for debugging
    print(f"Check: exists={instance_exists}, alive={instance_alive}")
    
    # Phase 4: Decision
    if instance_exists and instance_alive and instance:
        # ✅ Window tồn tại VÀ còn sống → Reuse
        print("✓ Instance already exists, bringing to front")
        instance.lift()
        instance.focus_force()
        return  # EXIT - Don't create new
